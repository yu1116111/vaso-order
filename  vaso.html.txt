function initLoginScreen(){
  const grid = document.getElementById('storeGrid');
  grid.innerHTML = STORES.map(s=>
    `<button class="store-btn" id="storeBtn_${s.id}" onclick="selectStore('${s.id}')">${s.name}</button>`
  ).join('');
  document.getElementById('loadingScreen').classList.add('hidden');
  document.getElementById('loginScreen').classList.remove('hidden');

  // 若之前登入過，嘗試自動登入
  const saved = localStorage.getItem('vaso_store');
  if(saved && STORES.find(s=>s.id===saved)){
    currentStore = saved;
    enterMainScreen();
  }
}
function selectStore(id){
  selectedStoreForLogin = id;
  STORES.forEach(s=>{
    document.getElementById('storeBtn_'+s.id).classList.toggle('active', s.id===id);
  });
}
function attemptLogin(){
  const err = document.getElementById('loginError');
  if(!selectedStoreForLogin){ err.textContent = '請先選擇分店'; return; }
  const store = STORES.find(s=>s.id===selectedStoreForLogin);
  const pw = document.getElementById('pwInput').value;
  if(pw !== store.password){ err.textContent = '密碼錯誤'; return; }
  err.textContent = '';
  currentStore = store.id;
  localStorage.setItem('vaso_store', store.id);
  enterMainScreen();
}
function logout(){
  localStorage.removeItem('vaso_store');
  location.reload();
}

function enterMainScreen(){
  document.getElementById('loginScreen').classList.add('hidden');
  document.getElementById('mainScreen').classList.remove('hidden');
  const storeName = STORES.find(s=>s.id===currentStore).name;
  document.getElementById('topbarStore').textContent = storeName + '店';
  renderDayPills();
  listenVendors();
}

function renderVendorList(){
  const el = document.getElementById('vendorList');
  const ids = Object.keys(vendors);

  // 篩選：非管理模式時只顯示今天要叫貨的廠商
  let showIds = ids;
  if(!adminMode){
    showIds = ids.filter(id=>{
      const v = vendors[id];
      return v.schedule && v.schedule[currentDay]==='order';
    });
  }

  let doneCount=0;
  showIds.forEach(id=>{ if(checkedVendors[id]) doneCount++; });
  document.getElementById('doneCount').textContent = doneCount;
  document.getElementById('totalCount').textContent = showIds.length;
  document.getElementById('progressFill').style.width = showIds.length ? (doneCount/showIds.length*100)+'%' : '0%';

  if(showIds.length===0){
    el.innerHTML = `<div style="text-align:center;color:var(--muted);padding:40px 20px;font-size:13px;">
      ${adminMode ? '目前沒有廠商，點上方「＋新增廠商」開始建立' : '今天沒有廠商要叫貨'}
    </div>`;
    return;
  }

  el.innerHTML = showIds.map(id=>renderVendorCard(id)).join('');
}

function scheduleLabel(v){
  const parts = [];
  DAY_NAMES.forEach((dn,idx)=>{
    const s = v.schedule && v.schedule[idx];
    if(s==='order') parts.push(dn+'叫');
    if(s==='deliver') parts.push(dn+'送');
  });
  return parts.join('、');
}

function renderVendorCard(id){
  const v = vendors[id];
  const isOpen = openVendorId===id;
  const isChecked = !!checkedVendors[id];
  const tool = openTool[id] || (v.hasOrderList ? 'order' : 'calc');

  let adminActions = '';
  if(adminMode){
    adminActions = `<div class="admin-row-actions">
      <button class="admin-edit-btn" onclick="event.stopPropagation();openVendorEditor('${id}')">編輯廠商</button>
      <button class="admin-del-btn" onclick="event.stopPropagation();deleteVendor('${id}')">刪除廠商</button>
    </div>`;
  }

  let detailHtml = '';
  if(isOpen){
    let tabsHtml = '';
    if(v.hasOrderList && v.hasCalculator){
      tabsHtml = `<div class="tool-tabs">
        <div class="tool-tab ${tool==='calc'?'active':''}" onclick="setTool('${id}','calc')">計算機</div>
        <div class="tool-tab ${tool==='order'?'active':''}" onclick="setTool('${id}','order')">叫貨單</div>
      </div>`;
    }
    let bodyHtml = '';
    if(tool==='order' && v.hasOrderList){
      bodyHtml = renderOrderList(id);
    } else if(tool==='calc' && v.hasCalculator){
      bodyHtml = renderCalculator(id);
    } else if(!v.hasOrderList && !v.hasCalculator){
      bodyHtml = `<div style="text-align:center;color:var(--muted);padding:16px;font-size:12.5px;">
        ${adminMode ? '這家廠商還沒啟用叫貨單或計算機，點「編輯廠商」開啟' : '這家廠商尚未設定叫貨單或計算機'}
      </div>`;
    }
    detailHtml = `<div class="vendor-detail">${tabsHtml}${bodyHtml}${adminActions}</div>`;
  }

  return `<div class="vendor-card ${isChecked?'done':''} ${isOpen?'open':''}">
    <div class="vendor-head" onclick="toggleVendorOpen('${id}')">
      <div class="vendor-circle ${isChecked?'checked':''}" onclick="event.stopPropagation();toggleVendorCheck('${id}')">✓</div>
      <div class="vendor-info">
        <div class="vendor-name">${esc(v.name)}</div>
        <div class="vendor-meta">${esc(scheduleLabel(v))}${v.note?' ・ '+esc(v.note):''}</div>
        ${v.payment?`<div class="vendor-payment">${esc(v.payment)}</div>`:''}
      </div>
      <div class="vendor-tag">叫貨</div>
      <div class="vendor-chevron">▾</div>
    </div>
    ${detailHtml}
  </div>`;
}

function esc(s){
  if(s==null) return '';
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
}

function toggleVendorOpen(id){
  openVendorId = (openVendorId===id) ? null : id;
  renderVendorList();
}
function toggleVendorCheck(id){
  checkedVendors[id] = !checkedVendors[id];
  renderVendorList();
}
function setTool(id,tool){
  openTool[id] = tool;
  renderVendorList();
}

function ensureOrderState(vid){
  if(!orderQty[vid]) orderQty[vid]={};
  if(!orderUnitIdx[vid]) orderUnitIdx[vid]={};
  const v = vendors[vid];
  if(v.orderList && v.orderList.groups){
    v.orderList.groups.forEach(g=>{
      (g.rows||[]).forEach(r=>{
        if(!(r.id in orderQty[vid])) orderQty[vid][r.id]=0;
      });
    });
  }
}

          unitHtml = `<button class="ol-unit-btn" onclick="toggleItemUnit('${vid}','${r.id}')">${esc(curUnit)}</button>`;
        } else {
          unitHtml = `<span class="ol-unit">${esc(r.unit||'')}</span>`;
        }
        ctrlHtml = `<button class="ol-btn" onclick="adjustQty('${vid}','${r.id}',-1)">−</button>
          <span class="ol-qv">${qStr}</span>${unitHtml}
          <button class="ol-btn plus" onclick="adjustQty('${vid}','${r.id}',1)">＋</button>`;
      }

      let adminRowActions='';
      if(adminMode){
        adminRowActions = `<div class="admin-row-actions">
          <button class="admin-edit-btn" onclick="openItemEditor('${vid}',{gi:${gi},ri:${ri}})">編輯</button>
          <button class="admin-del-btn" onclick="deleteItem('${vid}',${gi},${ri})">刪除</button>
        </div>`;
      }

      html += `<div class="ol-row ${isOn?'on':''}" style="flex-wrap:wrap;">
        <div class="ol-info">
          <div class="ol-name">${esc(r.name)}</div>
          ${r.note?`<div class="ol-note">${esc(r.note)}</div>`:''}
          ${adminRowActions}
        </div>
        <div class="ol-ctrl">${ctrlHtml}</div>
      </div>`;
    });
  });

  if(adminMode){
    html += `<button class="admin-add-btn" style="width:100%;margin-top:4px;" onclick="openItemEditor('${vid}',null)">＋新增品項</button>`;
  }

  // 匯出區
  const text = exportOrderText(vid);
  const lineCount = text ? text.split('\n').filter(l=>l && l!=='仁武店' && !l.includes('店')).length : 0;
  html += `<div class="ol-export">
    <div class="ol-export-head">
      <div>
        <div class="ol-export-title">匯出叫貨單</div>
        <div class="ol-export-cnt">${text?'已選擇品項':'尚未選擇任何品項'}</div>
      </div>
      <button class="ol-export-btn" onclick="copyOrderText('${vid}')">複製</button>
    </div>
    <div class="ol-export-text">${esc(text)||'（尚無內容）'}</div>
  </div>`;
  return html;
}

function adjustQty(vid,itemId,delta){
  ensureOrderState(vid);
  const row = findItemRow(vid,itemId);
  if(!row) return;
  let step = row.step||1;
  if(row.unit2 && row.stepUnit2!=null){
    const idx = orderUnitIdx[vid][itemId]||0;
    if(idx===1) step = row.stepUnit2;
  }
  const cur = orderQty[vid][itemId]||0;
  orderQty[vid][itemId] = Math.max(0, Math.round((cur+step*delta)*100)/100);
  renderVendorList();
}
function setPackQty(vid,itemId,packs,perPack){
  ensureOrderState(vid);
  orderQty[vid][itemId] = Math.round(parseFloat(packs)*parseFloat(perPack)*100)/100;
  renderVendorList();
}
function toggleItemUnit(vid,itemId){
  ensureOrderState(vid);
  const row = findItemRow(vid,itemId);
  if(!row || !row.unit2) return;
  const curIdx = orderUnitIdx[vid][itemId]||0;
  const newIdx = curIdx===0?1:0;
  // 換算數量
  const q = orderQty[vid][itemId]||0;
  const ratio = row.unitRatio||1; // 1 主單位 = ratio 副單位
  if(q>0){
    if(newIdx===1){ orderQty[vid][itemId] = Math.round(q*ratio*100)/100; }
    else { orderQty[vid][itemId] = Math.round(q/ratio*100)/100; }
  }
  orderUnitIdx[vid][itemId] = newIdx;
  renderVendorList();
}
function findItemRow(vid,itemId){
  const v = vendors[vid];
  if(!v.orderList) return null;
  let found=null;
  (v.orderList.groups||[]).forEach(g=>(g.rows||[]).forEach(r=>{ if(r.id===itemId) found=r; }));
  return found;
}
function exportOrderText(vid){
  const v = vendors[vid];
  if(!v.orderList) return '';
  const lines=[];
  (v.orderList.groups||[]).forEach(g=>(g.rows||[]).forEach(r=>{
    const q = orderQty[vid][r.id]||0;
    if(q>0){
      const qStr = (q%1===0)?String(q):String(Math.round(q*100)/100);
      const idx = orderUnitIdx[vid][r.id]||0;
      const unit = (idx===1 && r.unit2) ? r.unit2 : r.unit;
      let suffix='';
      const perPackVal = r.perPack || (r.packSelect ? r.packSelect.perPack : null);
      if(perPackVal && Math.abs(q-perPackVal) > 0.001){
        suffix = ` （${perPackVal}${unit} 1包）`;
      } else if(r.suffix && !perPackVal){
        suffix = ' '+r.suffix;
      }
      lines.push(`${r.name} ${qStr} ${unit}${suffix}`);
    }
  }));
  if(lines.length===0) return '';
  const storeName = STORES.find(s=>s.id===currentStore).name;
  let out = storeName+'店\n'+lines.join('\n');
  if(v.orderList.footer) out += '\n\n'+v.orderList.footer;
  return out;
}
function copyOrderText(vid){
  const text = exportOrderText(vid);
  if(!text){ alert('尚未選擇任何品項'); return; }
  navigator.clipboard.writeText(text).then(()=>{
    alert('已複製！可以貼到 LINE 了');
  }).catch(()=>{
    alert('複製失敗，請手動選取文字複製');
  });
}

function renderCalculator(vid){
  const v = vendors[vid];
  const calc = v.calculator;
  if(!calc || !calc.ingredients || calc.ingredients.length===0){
    let html = `<div style="text-align:center;color:var(--muted);padding:12px;font-size:12.5px;">
      ${adminMode?'尚未設定任何材料':'尚無計算機內容'}
    </div>`;
    if(adminMode){
      html += `<button class="admin-add-btn" style="width:100%;margin-top:8px;" onclick="openIngredientEditor('${vid}',null)">＋新增材料</button>`;
    }
    return html;
  }
  if(!calcChecked[vid]) calcChecked[vid]={};

  let html='';
  // 依分組顯示製品
  const groups = {};
  (calc.products||[]).forEach((p,pi)=>{
    const g = p.group||'其他';
    if(!groups[g]) groups[g]=[];
    groups[g].push({p,pi});
  });
  Object.keys(groups).forEach(gname=>{
    html += `<div class="calc-group-title">${esc(gname)}</div>`;
    groups[gname].forEach(({p,pi})=>{
      const checked = !!calcChecked[vid][p.id];
      const detail = Object.entries(p.usage||{}).map(([ing,amt])=>{
        const ingDef = calc.ingredients.find(i=>i.id===ing);
        return `${ingDef?ingDef.name:ing}${amt}`;
      }).join('・');
      let adminActions='';
      if(adminMode){
        adminActions = `<div class="admin-row-actions">
          <button class="admin-edit-btn" onclick="openProductEditor('${vid}',${pi})">編輯</button>
          <button class="admin-del-btn" onclick="deleteProduct('${vid}',${pi})">刪除</button>
        </div>`;
      }
      html += `<div class="calc-item">
        <div class="calc-cbox ${checked?'checked':''}" onclick="toggleCalcCheck('${vid}','${p.id}')"></div>
        <div style="flex:1;">
          <div class="calc-iname">${esc(p.name)}</div>
          <div class="calc-idetail">${esc(detail)}</div>
          ${adminActions}
        </div>
      </div>`;
    });
  });

  if(adminMode){
    html += `<div style="display:flex;gap:8px;margin-top:10px;">
      <button class="admin-add-btn" style="flex:1;" onclick="openIngredientEditor('${vid}',null)">＋材料</button>
      <button class="admin-add-btn" style="flex:1;" onclick="openProductEditor('${vid}',null)">＋製品配方</button>
    </div>`;
  }

  // 加總結果
  const totals = {};
  (calc.products||[]).forEach(p=>{
    if(calcChecked[vid][p.id]){
      Object.entries(p.usage||{}).forEach(([ing,amt])=>{
        totals[ing] = (totals[ing]||0)+amt;
      });
    }
  });
  const totalKeys = Object.keys(totals);
  if(totalKeys.length>0){
    html += `<div class="calc-result">
      <div class="calc-result-head">需要叫的總量</div>`;
    totalKeys.forEach(ingId=>{
      const ing = calc.ingredients.find(i=>i.id===ingId);
      if(!ing) return;
      const raw = totals[ingId];
      const packs = (raw/ing.spec).toFixed(2);
      html += `<div class="calc-rrow">
        <div class="calc-rname">${esc(ing.name)}<span class="calc-rspec">${ing.spec}${esc(ing.u)}/${esc(ing.unit)}</span></div>
        <div class="calc-rvals"><span class="calc-rg">${raw}${esc(ing.u)}</span><span class="calc-rpack">${packs} ${esc(ing.unit)}</span></div>
      </div>`;
    });
    html += `</div>`;
  }
  return html;
}
function toggleCalcCheck(vid,pid){
  if(!calcChecked[vid]) calcChecked[vid]={};
  calcChecked[vid][pid] = !calcChecked[vid][pid];
  renderVendorList();
}

function openVendorEditor(vid){
  editingVendorId = vid;
  const v = vid ? vendors[vid] : {name:'',schedule:{},note:'',payment:'',hasOrderList:false,hasCalculator:false};
  document.getElementById('vendorModalTitle').textContent = vid?'編輯廠商':'新增廠商';

  document.getElementById('vf_note').value = v.note||'';
  document.getElementById('vf_payment').value = v.payment||'';
  document.getElementById('vf_hasOrderList').checked = !!v.hasOrderList;
  document.getElementById('vf_hasCalculator').checked = !!v.hasCalculator;

  const sched = v.schedule||{};
  const grid = document.getElementById('vf_schedule');
  grid.innerHTML = DAY_NAMES.map((dn,idx)=>{
    const s = sched[idx]||'';
    return `<div class="day-check ${s}" data-day="${idx}" onclick="cycleScheduleDay(this)">${dn}</div>`;
  }).join('');

  openModal('vendorModal');
}
function cycleScheduleDay(el){
  const states = ['', 'order', 'deliver'];
  const cur = states.findIndex(s=>el.classList.contains(s) && s!=='') ;
  let curState = '';
  if(el.classList.contains('order')) curState='order';
  else if(el.classList.contains('deliver')) curState='deliver';
  const idx = states.indexOf(curState);
  const next = states[(idx+1)%states.length];
  el.classList.remove('order','deliver');
  if(next) el.classList.add(next);
}
function saveVendor(){
  const name = document.getElementById('vf_name').value.trim();
  if(!name){ alert('請輸入廠商名稱'); return; }
  const schedule = {};
  document.querySelectorAll('#vf_schedule .day-check').forEach(el=>{
    const day = el.dataset.day;
    if(el.classList.contains('order')) schedule[day]='order';
    else if(el.classList.contains('deliver')) schedule[day]='deliver';
  });
  const data = {
    name,
    schedule,
    note: document.getElementById('vf_note').value.trim(),
    payment: document.getElementById('vf_payment').value.trim(),
    hasOrderList: document.getElementById('vf_hasOrderList').checked,
    hasCalculator: document.getElementById('vf_hasCalculator').checked,
  };
  const id = editingVendorId || genId('v');
  if(editingVendorId && vendors[editingVendorId]){
    // 保留既有的 orderList / calculator 資料
    if(vendors[editingVendorId].orderList) data.orderList = vendors[editingVendorId].orderList;
    if(vendors[editingVendorId].calculator) data.calculator = vendors[editingVendorId].calculator;
  }
  vendorRef(id).set(data).then(()=>{
    closeModal('vendorModal');
  }).catch(e=>alert('儲存失敗：'+e.message));
}

function saveItem(){
  const {vid, gi, ri} = editingItemCtx;
  const groupTitle = document.getElementById('if_group').value.trim()||'品項';
  const name = document.getElementById('if_name').value.trim();
  if(!name){ alert('請輸入品項名稱'); return; }
  const row = {
    id: (gi!=null && ri!=null) ? vendors[vid].orderList.groups[gi].rows[ri].id : genId('item'),
    name,
    unit: document.getElementById('if_unit').value.trim()||'個',
    step: parseFloat(document.getElementById('if_step').value)||1,
    note: document.getElementById('if_note').value.trim(),
  };
  if(document.getElementById('if_useUnitSwitch').checked){
    row.unit2 = document.getElementById('if_unit2').value.trim();
    row.unitRatio = parseFloat(document.getElementById('if_ratio').value)||1;
  }
  if(document.getElementById('if_usePackSelect').checked){
    row.packSelect = {
      perPack: parseFloat(document.getElementById('if_perPack').value)||1,
      maxPacks: parseInt(document.getElementById('if_maxPacks').value)||20
    };
    row.perPack = row.packSelect.perPack; // 給匯出時判斷整包用
  }

  const v = JSON.parse(JSON.stringify(vendors[vid])); // deep clone
  if(!v.orderList) v.orderList = {groups:[]};
  if(gi!=null && ri!=null){
    // 編輯既有品項：分組若改名，搬到對應/新分組
    v.orderList.groups[gi].rows.splice(ri,1);
    if(v.orderList.groups[gi].rows.length===0) v.orderList.groups.splice(gi,1);
  }
  let targetGroup = v.orderList.groups.find(g=>g.title===groupTitle);
  if(!targetGroup){
    targetGroup = {title:groupTitle, rows:[]};
    v.orderList.groups.push(targetGroup);
  }
  targetGroup.rows.push(row);

  vendorRef(vid).child('orderList').set(v.orderList).then(()=>{
    closeModal('itemModal');
  }).catch(e=>alert('儲存失敗：'+e.message));
}
